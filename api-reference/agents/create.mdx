---
title: "Create Agent"
openapi: "POST /api/v1/agents/"
description: "Create a new voice AI agent with customizable voice, LLM model, telephony settings, and tool integrations."
---

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [Ravan.ai Dashboard](https://app.ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Body

<ParamField body="widget_settings_id" type="string">
  UUID of the widget settings to associate with this agent. Create widget settings first via the [Widget Settings API](/api-reference/widget-settings/create).
</ParamField>

<ParamField body="name" type="string">
  Internal name for the agent, visible in your dashboard.
</ParamField>

<ParamField body="agent_name" type="string">
  Display name shown to end users during calls and in the widget.
</ParamField>

<ParamField body="model" type="string">
  LLM model identifier for agent conversation (e.g. `gpt-4o`, `gpt-4o-mini`).
</ParamField>

<ParamField body="s2s_model" type="string">
  Speech-to-speech model identifier for real-time voice processing.
</ParamField>

<ParamField body="voice_id" type="string">
  Voice profile ID for text-to-speech output (e.g. `alloy`, `nova`, `shimmer`).
</ParamField>

<ParamField body="temperature" type="number">
  LLM temperature from `0.0` to `1.0`. Lower values produce more focused, deterministic responses. Default: `0.7`.
</ParamField>

<ParamField body="reminder_trigger_ms" type="integer">
  Milliseconds of silence before the agent sends a follow-up reminder (e.g. `8000` for 8 seconds).
</ParamField>

<ParamField body="reminder_max_count" type="integer">
  Maximum number of reminders the agent sends per call before stopping.
</ParamField>

<ParamField body="ambient_sound" type="string">
  Background ambient sound to play during calls (e.g. `office`, `cafe`).
</ParamField>

<ParamField body="ambient_sound_volume" type="number">
  Volume level for ambient sound, from `0.0` (silent) to `1.0` (full volume).
</ParamField>

<ParamField body="end_call_after_silence" type="boolean">
  If `true`, the agent automatically ends the call after prolonged user silence.
</ParamField>

<ParamField body="max_call_duration_ms" type="integer">
  Maximum call duration in milliseconds. The call auto-disconnects after this limit (e.g. `600000` for 10 minutes).
</ParamField>

<ParamField body="ring_duration_ms" type="integer">
  How long to ring before marking the call as unanswered (in milliseconds).
</ParamField>

<ParamField body="enable_voicemail_detection" type="boolean">
  If `true`, the agent detects voicemail greetings and leaves a message instead of continuing the conversation.
</ParamField>

<ParamField body="voicemail_message" type="string">
  The message the agent speaks when voicemail is detected.
</ParamField>

<ParamField body="voicemail_detection_timeout_ms" type="integer">
  Milliseconds to wait for speech before classifying the call as voicemail.
</ParamField>

<ParamField body="ivr_options" type="object">
  IVR (Interactive Voice Response) menu configuration for inbound call routing.
</ParamField>

<ParamField body="post_call_analysis_model" type="string">
  LLM model used for post-call analysis (e.g. `gpt-4o`).
</ParamField>

<ParamField body="analysis_summary_prompt" type="string">
  Custom prompt for generating the post-call summary and analysis.
</ParamField>

<ParamField body="selected_tools" type="object">
  Tool IDs to enable for this agent. Tools must be created first via the [Agent Tool API](/api-reference/tools/create).
</ParamField>

<ParamField body="integrations" type="object">
  Third-party integration configuration (e.g. calendar IDs, CRM references).
</ParamField>

<ParamField body="knowledge_base" type="string">
  Knowledge base content or reference that gives the agent domain-specific context.
</ParamField>

<ParamField body="begin_message" type="string">
  Opening greeting the agent speaks when the call starts.
</ParamField>

<ParamField body="start_speaker" type="string">
  Who speaks first: `"agent"` (agent greets the user) or `"user"` (agent waits for the user).
</ParamField>

---

### Response

<ResponseField name="success" type="boolean">Whether the request succeeded.</ResponseField>
<ResponseField name="message" type="string">Human-readable status message.</ResponseField>
<ResponseField name="data" type="object">
  The newly created agent object.

  <Expandable title="data">
    <ResponseField name="id" type="string">Unique agent ID (UUID) assigned by the server.</ResponseField>
    <ResponseField name="organization_id" type="string">ID of the organization that owns this agent.</ResponseField>
    <ResponseField name="widget_settings_id" type="string">ID of the widget settings associated with this agent.</ResponseField>
    <ResponseField name="name" type="string">Internal name used to identify the agent.</ResponseField>
    <ResponseField name="agent_name" type="string">Display name shown to end users.</ResponseField>
    <ResponseField name="status" type="string">Agent status — newly created agents default to `AGENT_STATUS_ACTIVE`.</ResponseField>
    <ResponseField name="model" type="string">LLM model identifier.</ResponseField>
    <ResponseField name="temperature" type="number">LLM temperature setting.</ResponseField>
    <ResponseField name="selected_tools" type="object">Tools enabled for this agent.</ResponseField>
    <ResponseField name="integrations" type="object">Integration configuration.</ResponseField>
    <ResponseField name="begin_message" type="string">Opening message spoken by the agent.</ResponseField>
    <ResponseField name="start_speaker" type="string">Who speaks first: `"agent"` or `"user"`.</ResponseField>
    <ResponseField name="created_at" type="string">ISO 8601 timestamp when the agent was created.</ResponseField>
    <ResponseField name="updated_at" type="string">ISO 8601 timestamp when the agent was last modified.</ResponseField>
  </Expandable>
</ResponseField>
