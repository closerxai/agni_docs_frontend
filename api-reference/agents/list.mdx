---
title: "List Agents"
openapi: "GET /api/v1/agents/"
description: "Retrieve a paginated list of all voice AI agents in your organization."
---

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [Ravan.ai Dashboard](https://app.ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Query Parameters

<ParamField query="limit" type="integer">
  Maximum number of agents to return per page. Defaults to all agents if not specified.
</ParamField>

<ParamField query="offset" type="integer">
  Number of records to skip for pagination. Defaults to `0`.
</ParamField>

---

### Response

<ResponseField name="success" type="boolean">
  Whether the request succeeded.
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable status message.
</ResponseField>

<ResponseField name="data" type="object[]">
  Array of agent objects.

  <Expandable title="data[]">
    <ResponseField name="id" type="string">Unique agent ID (UUID).</ResponseField>
    <ResponseField name="organization_id" type="string">ID of the organization that owns this agent.</ResponseField>
    <ResponseField name="widget_settings_id" type="string">ID of the widget settings associated with this agent.</ResponseField>
    <ResponseField name="name" type="string">Internal name used to identify the agent in your dashboard.</ResponseField>
    <ResponseField name="agent_name" type="string">Display name shown to end users during calls.</ResponseField>
    <ResponseField name="status" type="string">Current agent status. Allowed values: `AGENT_STATUS_UNSPECIFIED`, `AGENT_STATUS_ACTIVE`, `AGENT_STATUS_INACTIVE`.</ResponseField>
    <ResponseField name="model" type="string">LLM model identifier used for conversation (e.g. `gpt-4o`, `gpt-4o-mini`).</ResponseField>
    <ResponseField name="s2s_model" type="string">Speech-to-speech model identifier for real-time voice processing.</ResponseField>
    <ResponseField name="voice_id" type="string">Voice profile ID used for text-to-speech output.</ResponseField>
    <ResponseField name="temperature" type="number">LLM temperature (0.0–1.0). Lower values produce more deterministic responses.</ResponseField>
    <ResponseField name="reminder_trigger_ms" type="integer">Milliseconds of silence before the agent sends a reminder prompt.</ResponseField>
    <ResponseField name="reminder_max_count" type="integer">Maximum number of reminder prompts the agent will send per call.</ResponseField>
    <ResponseField name="ambient_sound" type="string">Background ambient sound identifier (e.g. `office`, `cafe`).</ResponseField>
    <ResponseField name="ambient_sound_volume" type="number">Volume level for ambient sound, from `0.0` (silent) to `1.0` (full).</ResponseField>
    <ResponseField name="end_call_after_silence" type="boolean">Whether to automatically end the call after prolonged silence.</ResponseField>
    <ResponseField name="max_call_duration_ms" type="integer">Maximum allowed call duration in milliseconds before auto-disconnect.</ResponseField>
    <ResponseField name="ring_duration_ms" type="integer">How long the phone rings (in ms) before the call is considered unanswered.</ResponseField>
    <ResponseField name="enable_voicemail_detection" type="boolean">Whether the agent should detect voicemail greetings and respond accordingly.</ResponseField>
    <ResponseField name="voicemail_message" type="string">Message the agent leaves when voicemail is detected.</ResponseField>
    <ResponseField name="voicemail_detection_timeout_ms" type="integer">Milliseconds to wait before concluding the call reached voicemail.</ResponseField>
    <ResponseField name="ivr_options" type="object">Interactive Voice Response (IVR) menu configuration for inbound call routing.</ResponseField>
    <ResponseField name="post_call_analysis_model" type="string">LLM model used to generate post-call analysis and summaries.</ResponseField>
    <ResponseField name="analysis_summary_prompt" type="string">Custom prompt template used for generating the post-call summary.</ResponseField>
    <ResponseField name="selected_tools" type="object">Map of tool IDs enabled for this agent during calls.</ResponseField>
    <ResponseField name="integrations" type="object">Third-party integration configuration (e.g. GHL calendar ID, webhook URLs).</ResponseField>
    <ResponseField name="knowledge_base" type="string">Knowledge base content or reference ID that provides context to the agent.</ResponseField>
    <ResponseField name="begin_message" type="string">Opening greeting message the agent speaks at the start of each call.</ResponseField>
    <ResponseField name="start_speaker" type="string">Who speaks first when the call connects — `"agent"` or `"user"`.</ResponseField>
    <ResponseField name="created_at" type="string">ISO 8601 timestamp when the agent was created.</ResponseField>
    <ResponseField name="updated_at" type="string">ISO 8601 timestamp when the agent was last modified.</ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="meta" type="object">
  Pagination metadata.

  <Expandable title="meta">
    <ResponseField name="total" type="integer">Total number of agents in your organization.</ResponseField>
    <ResponseField name="limit" type="integer">Page size used for this request.</ResponseField>
    <ResponseField name="offset" type="integer">Number of records skipped.</ResponseField>
  </Expandable>
</ResponseField>
