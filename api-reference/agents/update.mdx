---
title: "Update Agent"
openapi: "PATCH /api/v1/agents/{id}/"
description: "Update one or more fields of an existing voice AI agent. Only include the fields you want to change — unspecified fields remain unchanged."
---

<Note>
  This is a partial update (PATCH). Only include the fields you want to change in the request body.
</Note>

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [Ravan.ai Dashboard](https://app.ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Path Parameters

<ParamField path="id" type="string" required>
  The unique UUID of the agent to update.
</ParamField>

---

### Body

<ParamField body="widget_settings_id" type="string">UUID of the widget settings to associate.</ParamField>
<ParamField body="name" type="string">Internal name for the agent.</ParamField>
<ParamField body="agent_name" type="string">Display name shown to end users.</ParamField>
<ParamField body="model" type="string">LLM model identifier (e.g. `gpt-4o`, `gpt-4o-mini`).</ParamField>
<ParamField body="s2s_model" type="string">Speech-to-speech model identifier.</ParamField>
<ParamField body="voice_id" type="string">Voice profile ID for text-to-speech.</ParamField>
<ParamField body="temperature" type="number">LLM temperature (0.0–1.0).</ParamField>
<ParamField body="reminder_trigger_ms" type="integer">Milliseconds of silence before sending a reminder.</ParamField>
<ParamField body="reminder_max_count" type="integer">Maximum reminders per call.</ParamField>
<ParamField body="ambient_sound" type="string">Ambient sound identifier.</ParamField>
<ParamField body="ambient_sound_volume" type="number">Ambient sound volume (0.0–1.0).</ParamField>
<ParamField body="end_call_after_silence" type="boolean">Whether to end the call after prolonged silence.</ParamField>
<ParamField body="max_call_duration_ms" type="integer">Maximum call duration in milliseconds.</ParamField>
<ParamField body="ring_duration_ms" type="integer">Ring duration in milliseconds.</ParamField>
<ParamField body="enable_voicemail_detection" type="boolean">Enable or disable voicemail detection.</ParamField>
<ParamField body="voicemail_message" type="string">Message to leave on voicemail.</ParamField>
<ParamField body="voicemail_detection_timeout_ms" type="integer">Voicemail detection timeout in milliseconds.</ParamField>
<ParamField body="ivr_options" type="object">IVR configuration.</ParamField>
<ParamField body="post_call_analysis_model" type="string">Model for post-call analysis.</ParamField>
<ParamField body="analysis_summary_prompt" type="string">Prompt for post-call summary generation.</ParamField>
<ParamField body="selected_tools" type="object">Tool IDs enabled for this agent.</ParamField>
<ParamField body="integrations" type="object">Integration configuration.</ParamField>
<ParamField body="knowledge_base" type="string">Knowledge base content or reference.</ParamField>
<ParamField body="begin_message" type="string">Opening greeting message.</ParamField>
<ParamField body="start_speaker" type="string">Who speaks first: `"agent"` or `"user"`.</ParamField>

---

    <ResponseField name="temperature" type="number">LLM temperature.</ResponseField>

    <ResponseField name="begin_message" type="string">Opening message.</ResponseField>
