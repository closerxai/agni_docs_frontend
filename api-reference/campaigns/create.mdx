---
title: "Create Campaign"
description: "Create a new outbound voice campaign with contacts, scheduling rules, and retry configuration."
---

### Authorizations
<ParamField header="X-Api-Key" type="string" required>Your Agni API key.</ParamField>

---

### Body

<ParamField body="name" type="string" required>Campaign name (e.g. `"Q2 Outbound Blitz"`).</ParamField>
<ParamField body="agent_id" type="string" required>UUID of the AI agent that will handle calls for this campaign.</ParamField>
<ParamField body="phone_number_id" type="string" required>UUID of the phone number to use as the outbound line.</ParamField>
<ParamField body="from_phone_number" type="string" required>Caller ID in E.164 format (e.g. `+919876543210`). Must match the phone number.</ParamField>
<ParamField body="contact_ids" type="string[]" required>Array of contact UUIDs to include in the campaign.</ParamField>
<ParamField body="schedule" type="object">
  Optional scheduling configuration.

  <Expandable title="schedule">
    <ParamField body="schedule.window_start" type="string">Daily call window start time in 24h format (e.g. `"09:00"`).</ParamField>
    <ParamField body="schedule.window_end" type="string">Daily call window end time (e.g. `"18:00"`).</ParamField>
    <ParamField body="schedule.window_days" type="integer[]">Days of the week to run calls. `1`=Monday through `7`=Sunday (e.g. `[1,2,3,4,5]` for weekdays).</ParamField>
    <ParamField body="schedule.timezone" type="string">IANA timezone for the call window (e.g. `"Asia/Kolkata"`, `"America/New_York"`).</ParamField>
    <ParamField body="schedule.max_concurrent" type="integer">Maximum number of simultaneous outbound calls.</ParamField>
    <ParamField body="schedule.retry_attempts" type="integer">Maximum retry count per contact if the call fails or is unanswered.</ParamField>
    <ParamField body="schedule.retry_gap_min" type="integer">Minutes to wait between retry attempts.</ParamField>
  </Expandable>
</ParamField>

---

    <ResponseField name="execution_status" type="string">Execution state.</ResponseField>
    <ResponseField name="schedule" type="object">Scheduling configuration.</ResponseField>
    <ResponseField name="contact_stats" type="object">
      Aggregated contact counts.

      <Expandable title="contact_stats">

        <ResponseField name="no_answer" type="integer">No answer calls.</ResponseField>

  </Expandable>
</ResponseField>
