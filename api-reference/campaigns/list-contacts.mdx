---
title: "List Campaign Contacts"
description: "List contacts assigned to a campaign with their call status, attempt count, and results."
---

### Authorizations
<ParamField header="X-Api-Key" type="string" required>Your Agni API key.</ParamField>

### Path Parameters
<ParamField path="id" type="string" required>Campaign UUID.</ParamField>

### Query Parameters
<ParamField query="limit" type="integer">Max results per page.</ParamField>
<ParamField query="offset" type="integer">Pagination offset.</ParamField>

### Response
<ResponseField name="success" type="boolean">Whether the request succeeded.</ResponseField>
<ResponseField name="message" type="string">Human-readable status message.</ResponseField>
<ResponseField name="data" type="object[]">
  Array of campaign contact records.

  <Expandable title="data[]">
    <ResponseField name="id" type="string">Campaign-contact join UUID.</ResponseField>
    <ResponseField name="contact_id" type="string">Contact UUID.</ResponseField>
    <ResponseField name="name" type="string">Contact name.</ResponseField>
    <ResponseField name="phone" type="string">Contact phone number.</ResponseField>
    <ResponseField name="email" type="string">Contact email.</ResponseField>
    <ResponseField name="status" type="string">Call status. Values: `pending`, `in_progress`, `successful`, `failed`, `no_answer`.</ResponseField>
    <ResponseField name="attempt_count" type="integer">Number of call attempts made.</ResponseField>
    <ResponseField name="last_attempted_at" type="string">ISO 8601 timestamp of last attempt.</ResponseField>
    <ResponseField name="next_retry_at" type="string">ISO 8601 timestamp of next scheduled retry.</ResponseField>
    <ResponseField name="call_duration_sec" type="integer">Duration of the last call in seconds.</ResponseField>
    <ResponseField name="error_message" type="string">Error details if the call failed.</ResponseField>
    <ResponseField name="call_session_id" type="string">Session UUID for the call (links to call logs).</ResponseField>
    <ResponseField name="tags" type="string[]">Contact tags.</ResponseField>
  </Expandable>
</ResponseField>
<ResponseField name="meta" type="object">Pagination metadata with `total`, `limit`, `offset`.</ResponseField>
