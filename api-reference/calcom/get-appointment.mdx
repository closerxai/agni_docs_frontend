---
title: "Get Cal.com Appointment"
openapi: "GET /api/v1/calcom/appointments/{id}"
description: "Retrieve details of a single Cal.com appointment by its ID."
---

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [Ravan.ai Dashboard](https://app.ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Path Parameters

<ParamField path="id" type="string" required>
  The unique appointment ID to retrieve.
</ParamField>

---

### Query Parameters

<ParamField query="org_id" type="string" required>
  UUID of your organization.
</ParamField>

---

### Response

<ResponseField name="success" type="boolean">Whether the request succeeded.</ResponseField>
<ResponseField name="message" type="string">Human-readable status message.</ResponseField>
<ResponseField name="data" type="object">
  The appointment record.

  <Expandable title="data">
    <ResponseField name="id" type="string">Internal appointment record ID.</ResponseField>
    <ResponseField name="organization_id" type="string">Organization ID.</ResponseField>
    <ResponseField name="agent_id" type="string">ID of the agent that handled this booking.</ResponseField>
    <ResponseField name="calcom_booking_uid" type="string">Cal.com unique booking identifier.</ResponseField>
    <ResponseField name="attendee_name" type="string">Full name of the attendee.</ResponseField>
    <ResponseField name="attendee_email" type="string">Email address of the attendee.</ResponseField>
    <ResponseField name="attendee_phone" type="string">Phone number (E.164).</ResponseField>
    <ResponseField name="attendee_timezone" type="string">IANA timezone (e.g. `America/New_York`).</ResponseField>
    <ResponseField name="start_time" type="string">Appointment start time (ISO 8601).</ResponseField>
    <ResponseField name="end_time" type="string">Appointment end time (ISO 8601).</ResponseField>
    <ResponseField name="duration_minutes" type="integer">Duration in minutes.</ResponseField>
    <ResponseField name="meeting_url" type="string">Video meeting URL.</ResponseField>
    <ResponseField name="status" type="string">Booking status.</ResponseField>
  </Expandable>
</ResponseField>
