---
title: "Lookup Cal.com Event Type"
openapi: "GET /api/v1/calcom/event-types/lookup/{platform_id}"
description: "Look up a specific Cal.com event type by its platform ID or URL slug."
---

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [support@ravan.ai](mailto:support@ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Path Parameters

<ParamField path="platform_id" type="string" required>
  Cal.com event type ID or slug (e.g. `evt_123456` or `intro-call`).
</ParamField>

---

### Response

<ResponseField name="success" type="boolean">Whether the request succeeded.</ResponseField>
<ResponseField name="message" type="string">Human-readable status message.</ResponseField>
<ResponseField name="data" type="object">
  The event type details.

  <Expandable title="data">
    <ResponseField name="id" type="string">Cal.com event type ID.</ResponseField>
    <ResponseField name="slug" type="string">URL slug for the event type.</ResponseField>
    <ResponseField name="title" type="string">Display title.</ResponseField>
    <ResponseField name="description" type="string">Event type description.</ResponseField>
    <ResponseField name="duration_minutes" type="integer">Default duration in minutes.</ResponseField>
    <ResponseField name="price" type="number">Event price (0 if free).</ResponseField>
    <ResponseField name="currency" type="string">Currency code.</ResponseField>
    <ResponseField name="is_active" type="boolean">Whether the event type is active.</ResponseField>
  </Expandable>
</ResponseField>
