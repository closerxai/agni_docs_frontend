---
title: "List Cal.com Appointments"
openapi: "GET /api/v1/calcom/appointments"
description: "Retrieve appointments booked through Cal.com, including attendee details, meeting URLs, and booking status."
---

<Note>
  Requires an active Cal.com connection. Use the [Cal.com Status](/api-reference/calcom/status) endpoint to verify your connection.
</Note>

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [Ravan.ai Dashboard](https://app.ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Query Parameters

<ParamField query="status" type="string">Filter by booking status (e.g. `accepted`, `cancelled`, `pending`).</ParamField>
<ParamField query="agent_id" type="string">Filter appointments by agent ID.</ParamField>
<ParamField query="limit" type="integer">Maximum number of appointments to return (max: `200`).</ParamField>

---

### Response

<ResponseField name="success" type="boolean">Whether the request succeeded.</ResponseField>
<ResponseField name="message" type="string">Human-readable status message.</ResponseField>
<ResponseField name="data" type="object">
  Appointment results.

  <Expandable title="data">
    <ResponseField name="appointments" type="object[]">
      Array of appointment records.

      <Expandable title="appointments[]">
        <ResponseField name="id" type="string">Internal appointment record ID.</ResponseField>
        <ResponseField name="organization_id" type="string">Organization ID.</ResponseField>
        <ResponseField name="agent_id" type="string">ID of the agent that handled this booking.</ResponseField>
        <ResponseField name="calcom_booking_uid" type="string">Cal.com unique booking identifier.</ResponseField>
        <ResponseField name="attendee_name" type="string">Full name of the attendee.</ResponseField>
        <ResponseField name="attendee_email" type="string">Email address of the attendee.</ResponseField>
        <ResponseField name="attendee_phone" type="string">Phone number of the attendee in E.164 format.</ResponseField>
        <ResponseField name="attendee_timezone" type="string">IANA timezone of the attendee (e.g. `America/New_York`).</ResponseField>
        <ResponseField name="start_time" type="string">Appointment start time in ISO 8601.</ResponseField>
        <ResponseField name="end_time" type="string">Appointment end time in ISO 8601.</ResponseField>
        <ResponseField name="duration_minutes" type="integer">Duration of the appointment in minutes.</ResponseField>
        <ResponseField name="meeting_url" type="string">Video meeting URL (e.g. Google Meet, Zoom link).</ResponseField>
        <ResponseField name="status" type="string">Booking status (e.g. `accepted`, `cancelled`, `pending`).</ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="count" type="integer">Total number of appointments returned.</ResponseField>
  </Expandable>
</ResponseField>
