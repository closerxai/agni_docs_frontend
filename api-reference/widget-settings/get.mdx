---
title: "Get Widget Settings"
openapi: "GET /api/v1/widget-settings/{id}/"
description: "Retrieve the full configuration of a single widget settings object by its ID."
---

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [Ravan.ai Dashboard](https://app.ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Path Parameters

<ParamField path="id" type="string" required>
  The unique UUID of the widget settings to retrieve.
</ParamField>

---

### Response

<ResponseField name="success" type="boolean">
  Whether the request succeeded.
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable status message.
</ResponseField>

<ResponseField name="data" type="object">
  The widget settings object.

  <Expandable title="data">
    <ResponseField name="id" type="string">Unique widget settings ID (UUID).</ResponseField>
    <ResponseField name="bot_name" type="string">Display name of the bot shown in the widget header.</ResponseField>
    <ResponseField name="bot_auto_start" type="boolean">Whether the widget automatically starts a conversation on page load.</ResponseField>
    <ResponseField name="bot_position" type="string">Widget position. Values: `BOT_POSITION_BOTTOM_RIGHT`, `BOT_POSITION_BOTTOM_LEFT`, `BOT_POSITION_TOP_RIGHT`, `BOT_POSITION_TOP_LEFT`, `BOT_POSITION_UNSPECIFIED`.</ResponseField>
    <ResponseField name="bot_logo" type="string">URL of the bot avatar image.</ResponseField>
    <ResponseField name="svg_logo" type="string">Inline SVG markup for the widget icon button.</ResponseField>
    <ResponseField name="custom_form_fields" type="object">Custom form fields configuration.</ResponseField>
    <ResponseField name="theme" type="object">Widget color theme and styling.</ResponseField>
    <ResponseField name="language" type="string">Language code (e.g. `en`, `es`, `fr`).</ResponseField>
    <ResponseField name="consent_content" type="string">Consent modal body text.</ResponseField>
    <ResponseField name="consent_title" type="string">Consent modal title.</ResponseField>
    <ResponseField name="is_active" type="boolean">Whether these settings are active.</ResponseField>
    <ResponseField name="created_at" type="string">Creation timestamp (ISO 8601).</ResponseField>
    <ResponseField name="updated_at" type="string">Last update timestamp (ISO 8601).</ResponseField>
  </Expandable>
</ResponseField>
