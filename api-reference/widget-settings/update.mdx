---
title: "Update Widget Settings"
openapi: "PATCH /api/v1/widget-settings/{id}/"
description: "Update one or more fields of an existing widget configuration. Only include the fields you want to change."
---

<Note>
  This is a partial update (PATCH). Only include the fields you want to modify in the request body.
</Note>

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [Ravan.ai Dashboard](https://app.ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Path Parameters

<ParamField path="id" type="string" required>
  The unique UUID of the widget settings to update.
</ParamField>

---

### Body

<ParamField body="bot_name" type="string">Display name of the bot.</ParamField>
<ParamField body="bot_auto_start" type="boolean">Whether the widget auto-starts on page load.</ParamField>
<ParamField body="bot_position" type="string">Widget position. Values: `BOT_POSITION_BOTTOM_RIGHT`, `BOT_POSITION_BOTTOM_LEFT`, `BOT_POSITION_TOP_RIGHT`, `BOT_POSITION_TOP_LEFT`.</ParamField>
<ParamField body="bot_logo" type="string">URL of the bot avatar image.</ParamField>
<ParamField body="svg_logo" type="string">Inline SVG markup for the widget button.</ParamField>
<ParamField body="custom_form_fields" type="object">Custom form field configuration.</ParamField>
<ParamField body="theme" type="object">Widget theme and styling configuration.</ParamField>
<ParamField body="language" type="string">Language code (e.g. `en`, `es`).</ParamField>
<ParamField body="consent_content" type="string">Consent modal body text.</ParamField>
<ParamField body="consent_title" type="string">Consent modal title.</ParamField>

---

### Response

<ResponseField name="success" type="boolean">Whether the request succeeded.</ResponseField>
<ResponseField name="message" type="string">Human-readable status message.</ResponseField>
<ResponseField name="data" type="object">
  The updated widget settings object.

  <Expandable title="data">
    <ResponseField name="id" type="string">Widget settings ID (UUID).</ResponseField>
    <ResponseField name="bot_name" type="string">Display name of the bot.</ResponseField>
    <ResponseField name="bot_auto_start" type="boolean">Whether the widget auto-starts.</ResponseField>
    <ResponseField name="bot_position" type="string">Widget position on the page.</ResponseField>
    <ResponseField name="bot_logo" type="string">Bot avatar image URL.</ResponseField>
    <ResponseField name="svg_logo" type="string">SVG icon markup.</ResponseField>
    <ResponseField name="custom_form_fields" type="object">Custom form fields.</ResponseField>
    <ResponseField name="theme" type="object">Widget theme.</ResponseField>
    <ResponseField name="language" type="string">Language code.</ResponseField>
    <ResponseField name="consent_content" type="string">Consent modal body text.</ResponseField>
    <ResponseField name="consent_title" type="string">Consent modal title.</ResponseField>
    <ResponseField name="is_active" type="boolean">Whether the settings are active.</ResponseField>
    <ResponseField name="created_at" type="string">Creation timestamp (ISO 8601).</ResponseField>
    <ResponseField name="updated_at" type="string">Last update timestamp (ISO 8601).</ResponseField>
  </Expandable>
</ResponseField>
