---
title: "Create Widget Settings"
openapi: "POST /api/v1/widget-settings/"
description: "Create a new widget configuration that controls the appearance, position, and behavior of an embeddable voice chat widget on your website."
---

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [Ravan.ai Dashboard](https://app.ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Body

<ParamField body="bot_name" type="string">
  Display name of the bot shown in the widget header (e.g. `"Ava Support"`).
</ParamField>

<ParamField body="bot_auto_start" type="boolean">
  If `true`, the widget automatically starts a voice conversation when the page loads. Defaults to `false`.
</ParamField>

<ParamField body="bot_position" type="string">
  Where the widget appears on the page. Allowed values:
  - `BOT_POSITION_BOTTOM_RIGHT` (default)
  - `BOT_POSITION_BOTTOM_LEFT`
  - `BOT_POSITION_TOP_RIGHT`
  - `BOT_POSITION_TOP_LEFT`
  - `BOT_POSITION_UNSPECIFIED`
</ParamField>

<ParamField body="bot_logo" type="string">
  URL of the bot avatar image displayed inside the widget.
</ParamField>

<ParamField body="svg_logo" type="string">
  Inline SVG markup for the floating widget button icon.
</ParamField>

<ParamField body="custom_form_fields" type="object">
  Custom form fields displayed to the user before starting a conversation (e.g. name, email, phone number).
</ParamField>

<ParamField body="theme" type="object">
  Widget color theme and styling. Pass a theme configuration object to customize colors, fonts, and appearance.
</ParamField>

<ParamField body="language" type="string">
  Language code for widget UI text (e.g. `en`, `es`, `fr`, `de`).
</ParamField>

<ParamField body="consent_content" type="string">
  Body text of the consent/privacy modal shown to users before call recording begins.
</ParamField>

<ParamField body="consent_title" type="string">
  Title of the consent/privacy modal.
</ParamField>

---

### Response

<ResponseField name="success" type="boolean">
  Whether the request succeeded.
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable status message.
</ResponseField>

<ResponseField name="data" type="object">
  The newly created widget settings object.

  <Expandable title="data">
    <ResponseField name="id" type="string">Unique widget settings ID (UUID) assigned by the server.</ResponseField>
    <ResponseField name="bot_name" type="string">Display name of the bot.</ResponseField>
    <ResponseField name="bot_auto_start" type="boolean">Whether the widget auto-starts.</ResponseField>
    <ResponseField name="bot_position" type="string">Widget position on the page.</ResponseField>
    <ResponseField name="bot_logo" type="string">Bot avatar image URL.</ResponseField>
    <ResponseField name="svg_logo" type="string">SVG icon markup.</ResponseField>
    <ResponseField name="custom_form_fields" type="object">Custom form fields configuration.</ResponseField>
    <ResponseField name="theme" type="object">Widget theme configuration.</ResponseField>
    <ResponseField name="language" type="string">Language code.</ResponseField>
    <ResponseField name="consent_content" type="string">Consent modal body text.</ResponseField>
    <ResponseField name="consent_title" type="string">Consent modal title.</ResponseField>
    <ResponseField name="is_active" type="boolean">Whether the settings are active (defaults to `true`).</ResponseField>
    <ResponseField name="created_at" type="string">Creation timestamp (ISO 8601).</ResponseField>
    <ResponseField name="updated_at" type="string">Last update timestamp (ISO 8601).</ResponseField>
  </Expandable>
</ResponseField>

  -d '{

response = requests.post(

        "bot_position": "BOT_POSITION_BOTTOM_RIGHT",

print(response.json())
