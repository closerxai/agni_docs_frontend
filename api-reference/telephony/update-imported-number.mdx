---
title: "Update Imported Number"
openapi: "PUT /api/v1/phone-numbers/import/{id}"
description: "Update the SIP configuration or friendly name of an imported phone number."
---

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [support@ravan.ai](mailto:support@ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Path Parameters

<ParamField path="id" type="string" required>
  The unique ID of the imported phone number to update.
</ParamField>

---

### Body

<ParamField body="name" type="string">
  Updated friendly name for the number.
</ParamField>

<ParamField body="termination_uri" type="string">
  Updated SIP termination URI.
</ParamField>

<ParamField body="sip_username" type="string">
  Updated SIP authentication username.
</ParamField>

<ParamField body="sip_password" type="string">
  Updated SIP authentication password.
</ParamField>

---

### Response

<ResponseField name="success" type="boolean">Whether the request succeeded.</ResponseField>
<ResponseField name="message" type="string">Human-readable status message.</ResponseField>
<ResponseField name="data" type="object">
  The updated imported phone number record.

  <Expandable title="data">
    <ResponseField name="id" type="string">Unique record ID.</ResponseField>
    <ResponseField name="phone" type="string">The imported phone number in E.164 format.</ResponseField>
    <ResponseField name="type" type="string">Telephony connection type.</ResponseField>
    <ResponseField name="name" type="string">Updated friendly name.</ResponseField>
    <ResponseField name="termination_uri" type="string">Updated SIP termination URI.</ResponseField>
    <ResponseField name="sip_username" type="string">Updated SIP username.</ResponseField>
    <ResponseField name="sip_password" type="string">Updated SIP password.</ResponseField>
    <ResponseField name="status" type="string">Number status.</ResponseField>
    <ResponseField name="created_at" type="string">ISO 8601 creation timestamp.</ResponseField>
  </Expandable>
</ResponseField>
