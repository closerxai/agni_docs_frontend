---
title: "Import Phone Number"
openapi: "POST /api/v1/phone-numbers/import/"
description: "Import an existing phone number from your own SIP trunk or telephony provider into the Agni platform."
---

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [Ravan.ai Dashboard](https://app.ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Body

<ParamField body="phone" type="string" required>
  Phone number to import in E.164 format (e.g. `+14155550100`).
</ParamField>

<ParamField body="type" type="string" required>
  Type of telephony connection — e.g. `sip`, `pstn`.
</ParamField>

<ParamField body="name" type="string">
  Friendly name for the imported number (e.g. `"Office Main Line"`).
</ParamField>

<ParamField body="termination_uri" type="string">
  SIP termination URI for routing calls (e.g. `sip:trunk@provider.com`).
</ParamField>

<ParamField body="sip_username" type="string">
  SIP authentication username.
</ParamField>

<ParamField body="sip_password" type="string">
  SIP authentication password.
</ParamField>

---

### Response

<ResponseField name="success" type="boolean">Whether the request succeeded.</ResponseField>
<ResponseField name="message" type="string">Human-readable status message.</ResponseField>
<ResponseField name="data" type="object">
  The imported phone number record.

  <Expandable title="data">
    <ResponseField name="id" type="string">Unique record ID assigned by the server.</ResponseField>
    <ResponseField name="phone" type="string">The imported phone number in E.164 format.</ResponseField>
    <ResponseField name="type" type="string">Telephony connection type.</ResponseField>
    <ResponseField name="name" type="string">Friendly name for the number.</ResponseField>
    <ResponseField name="termination_uri" type="string">SIP termination URI.</ResponseField>
    <ResponseField name="sip_username" type="string">SIP username.</ResponseField>
    <ResponseField name="sip_password" type="string">SIP password (returned on creation only).</ResponseField>
    <ResponseField name="status" type="string">Number status (e.g. `active`).</ResponseField>
    <ResponseField name="created_at" type="string">ISO 8601 creation timestamp.</ResponseField>
  </Expandable>
</ResponseField>
