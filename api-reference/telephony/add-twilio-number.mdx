---
title: "Add Twilio Number"
openapi: "POST /api/v1/phone-numbers/add-twilio"
description: "Add a phone number from your connected Twilio account to the Agni platform for use with your voice agents."
---

<Note>
  You must first connect your Twilio account using [Create Telephony Type](/api-reference/telephony/create-telephony-type) and retrieve your numbers via [List Twilio Numbers](/api-reference/telephony/list-twilio-numbers) before adding them.
</Note>

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [support@ravan.ai](mailto:support@ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Body

<ParamField body="phone_number" type="string" required>
  Phone number in E.164 format (e.g. `+14155550100`). Must be a number from your Twilio account.
</ParamField>

<ParamField body="friendly_name" type="string">
  A friendly label for the number.
</ParamField>

<ParamField body="price" type="number">
  Monthly price of the number.
</ParamField>

<ParamField body="per_minute_price_inbound" type="number">
  Per-minute inbound call rate.
</ParamField>

<ParamField body="per_minute_price_outbound" type="number">
  Per-minute outbound call rate.
</ParamField>

<ParamField body="region" type="string">
  Region or area code of the number.
</ParamField>

<ParamField body="iso_country" type="string">
  ISO country code (e.g. `US`, `GB`).
</ParamField>

<ParamField body="number_sid" type="string" required>
  The Twilio SID for this phone number (starts with `PN`).
</ParamField>

---

### Response

<ResponseField name="success" type="boolean">Whether the request succeeded.</ResponseField>
<ResponseField name="message" type="string">Human-readable status message.</ResponseField>
<ResponseField name="data" type="object">
  The added phone number record.

  <Expandable title="data">
    <ResponseField name="id" type="string">Unique record ID in Agni.</ResponseField>
    <ResponseField name="telephony_id" type="string">Telephony configuration ID.</ResponseField>
    <ResponseField name="subscription_id" type="string">Subscription ID if applicable.</ResponseField>
    <ResponseField name="inbound_trunk_id" type="string">Inbound SIP trunk ID.</ResponseField>
    <ResponseField name="outbound_trunk_id" type="string">Outbound SIP trunk ID.</ResponseField>
    <ResponseField name="number" type="string">Phone number in E.164 format.</ResponseField>
    <ResponseField name="friendly_name" type="string">Display name.</ResponseField>
    <ResponseField name="type" type="string">Number type.</ResponseField>
    <ResponseField name="per_minute_price_inbound" type="number">Inbound per-minute rate.</ResponseField>
    <ResponseField name="per_minute_price_outbound" type="number">Outbound per-minute rate.</ResponseField>
    <ResponseField name="price" type="number">Monthly price.</ResponseField>
    <ResponseField name="region" type="string">Region.</ResponseField>
    <ResponseField name="iso_country" type="string">ISO country code.</ResponseField>
    <ResponseField name="is_active" type="boolean">Whether the number is active.</ResponseField>
    <ResponseField name="created_at" type="string">ISO 8601 creation timestamp.</ResponseField>
    <ResponseField name="updated_at" type="string">ISO 8601 last update timestamp.</ResponseField>
  </Expandable>
</ResponseField>
