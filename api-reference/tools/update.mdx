---
title: "Update Agent Tool"
description: "Update the name, description, type, or definition of an existing agent tool."
openapi: "PATCH /api/v1/tools/{id}/"
---

<Note>
  This is a partial update (PATCH). Only include the fields you want to change.
</Note>

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [support@ravan.ai](mailto:support@ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Path Parameters

<ParamField path="id" type="string" required>
  The unique UUID of the tool to update.
</ParamField>

---

### Body

<ParamField body="name" type="string">
  Updated tool name (lowercase, underscores, no spaces).
</ParamField>

<ParamField body="description" type="string">
  Updated description of what the tool does.
</ParamField>

<ParamField body="type" type="string">
  Updated tool type — `function` or `webhook`.
</ParamField>

<ParamField body="definition" type="object">
  Updated JSON Schema definition with parameter schema.
</ParamField>

---

### Response

<ResponseField name="success" type="boolean">
  Whether the request succeeded.
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable status message.
</ResponseField>

<ResponseField name="data" type="object">
  The updated tool object.

  <Expandable title="data">
    <ResponseField name="id" type="string">Unique tool ID (UUID).</ResponseField>
    <ResponseField name="organization_id" type="string">Organization ID.</ResponseField>
    <ResponseField name="name" type="string">Tool name.</ResponseField>
    <ResponseField name="description" type="string">Tool description.</ResponseField>
    <ResponseField name="type" type="string">Tool type.</ResponseField>
    <ResponseField name="definition" type="object">Tool definition schema.</ResponseField>
    <ResponseField name="is_active" type="boolean">Whether the tool is active.</ResponseField>
  </Expandable>
</ResponseField>
