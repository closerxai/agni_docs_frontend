---
title: "Get Agent Tool"
description: "Retrieve the full definition and status of a single agent tool by its ID."
---

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [support@ravan.ai](mailto:support@ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Path Parameters

<ParamField path="id" type="string" required>
  The unique UUID of the tool to retrieve.
</ParamField>

---

### Response

<ResponseField name="success" type="boolean">
  Whether the request succeeded.
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable status message.
</ResponseField>

<ResponseField name="data" type="object">
  The tool object.

  <Expandable title="data">
    <ResponseField name="id" type="string">
      Unique tool ID (UUID).
    </ResponseField>

    <ResponseField name="organization_id" type="string">
      ID of the organization that owns this tool.
    </ResponseField>

    <ResponseField name="name" type="string">
      Tool name (lowercase, underscores, no spaces).
    </ResponseField>

    <ResponseField name="description" type="string">
      Human-readable description shown to the LLM.
    </ResponseField>

    <ResponseField name="type" type="string">
      Tool type — e.g. `function`, `webhook`.
    </ResponseField>

    <ResponseField name="definition" type="object">
      JSON Schema definition including parameter schema.
    </ResponseField>

    <ResponseField name="is_active" type="boolean">
      Whether the tool is active.
    </ResponseField>
  </Expandable>
</ResponseField>
