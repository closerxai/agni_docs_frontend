---
title: "List Agent Tools"
openapi: "GET /api/v1/tools/"
description: "Retrieve a paginated list of all custom tools (functions, webhooks) available in your organization."
---

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [support@ravan.ai](mailto:support@ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Query Parameters

<ParamField query="limit" type="integer">
  Maximum number of tools to return per page.
</ParamField>

<ParamField query="offset" type="integer">
  Number of records to skip for pagination. Defaults to `0`.
</ParamField>

---

### Response

<ResponseField name="success" type="boolean">
  Whether the request succeeded.
</ResponseField>

<ResponseField name="message" type="string">
  Human-readable status message.
</ResponseField>

<ResponseField name="data" type="object[]">
  Array of tool objects.

  <Expandable title="data[]">
    <ResponseField name="id" type="string">
      Unique tool ID (UUID).
    </ResponseField>

    <ResponseField name="organization_id" type="string">
      ID of the organization that owns this tool.
    </ResponseField>

    <ResponseField name="name" type="string">
      Tool name. Must be a valid function name (lowercase, underscores, no spaces).
    </ResponseField>

    <ResponseField name="description" type="string">
      Human-readable description of what the tool does. This is shown to the LLM to decide when to invoke the tool.
    </ResponseField>

    <ResponseField name="type" type="string">
      Tool type — e.g. `function`, `webhook`.
    </ResponseField>

    <ResponseField name="definition" type="object">
      JSON Schema definition of the tool including its name, description, and parameter schema. Follows the OpenAI function calling format.
    </ResponseField>

    <ResponseField name="is_active" type="boolean">
      Whether the tool is currently active and available for agents to use.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="meta" type="object">
  Pagination metadata.

  <Expandable title="meta">
    <ResponseField name="total" type="integer">
      Total number of tools in your organization.
    </ResponseField>

    <ResponseField name="limit" type="integer">
      Page size used for this request.
    </ResponseField>

    <ResponseField name="offset" type="integer">
      Number of records skipped.
    </ResponseField>
  </Expandable>
</ResponseField>
