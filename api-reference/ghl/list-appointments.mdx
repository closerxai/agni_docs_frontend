---
title: "List GHL Appointments"
openapi: "GET /api/v1/ghl/appointments"
description: "Retrieve a list of appointments booked through the GoHighLevel integration, with contact details and status."
---

### Authorizations

<ParamField header="X-Api-Key" type="string" required>
  Your Agni API key. Find it in your [support@ravan.ai](mailto:support@ravan.ai) under **Settings → API Keys**.
</ParamField>

---

### Query Parameters

<ParamField query="agent_id" type="string">Filter appointments by agent ID.</ParamField>
<ParamField query="status" type="string">Filter by appointment status (e.g. `confirmed`, `cancelled`).</ParamField>
<ParamField query="limit" type="integer">Maximum number of appointments to return (max: `200`).</ParamField>
<ParamField query="org_id" type="string">Organization ID (optional fallback for authentication).</ParamField>

---

### Response

<ResponseField name="success" type="boolean">Whether the request succeeded.</ResponseField>
<ResponseField name="message" type="string">Human-readable status message.</ResponseField>
<ResponseField name="data" type="object">
  Appointment results.

  <Expandable title="data">
    <ResponseField name="appointments" type="object[]">
      Array of appointment records.

      <Expandable title="appointments[]">
        <ResponseField name="id" type="string">Internal appointment record ID.</ResponseField>
        <ResponseField name="ghl_appointment_id" type="string">GoHighLevel appointment ID.</ResponseField>
        <ResponseField name="contact_id" type="string">GHL contact ID of the customer.</ResponseField>
        <ResponseField name="agent_id" type="string">ID of the agent that handled this appointment.</ResponseField>
        <ResponseField name="agent_name" type="string">Name of the agent.</ResponseField>
        <ResponseField name="contact_name" type="string">Customer's full name.</ResponseField>
        <ResponseField name="contact_email" type="string">Customer's email address.</ResponseField>
        <ResponseField name="contact_phone" type="string">Customer's phone number in E.164 format.</ResponseField>
        <ResponseField name="appointment_time" type="string">Appointment date and time in ISO 8601 format.</ResponseField>
        <ResponseField name="status" type="string">Appointment status (e.g. `confirmed`, `cancelled`, `completed`).</ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="count" type="integer">Total number of appointments returned.</ResponseField>
  </Expandable>
</ResponseField>
